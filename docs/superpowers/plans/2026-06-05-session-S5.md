- [x] **Step 1.1: Add alpha-compositing + rule-extraction helpers.** Append to `check_dashboard_contrast.py` (keep all existing code):

```python
# --- S5: CSS-rule-aware + alpha-blend contrast checks ---------------------
def _rgb(hex_color):
    v = hex_color.lstrip("#")
    return tuple(int(v[i:i + 2], 16) for i in (0, 2, 4))

def _hex(rgb):
    return "#%02x%02x%02x" % tuple(int(round(c)) for c in rgb)

def composite(layers):
    """layers: [(hex, alpha), ...] painted bottom→top. First layer should be opaque."""
    base = _rgb(layers[0][0])
    for hex_color, alpha in layers[1:]:
        top = _rgb(hex_color)
        base = tuple(alpha * t + (1 - alpha) * b for t, b in zip(top, base))
    return _hex(base)

def resolve(token_or_hex, colors):
    """Resolve 'var(--x)' or '#hex' to a #hex using the parsed :root vars."""
    m = re.search(r"var\(\s*--([a-zA-Z0-9-]+)\s*\)", token_or_hex)
    if m:
        return colors.get(m.group(1))
    m = re.search(r"#[0-9a-fA-F]{6}", token_or_hex)
    return m.group(0) if m else None

def rule_decl(css, selector, prop):
    """Last value of `prop` in every rule block whose comma-split selector list
    contains `selector` exactly. Returns the raw value string or None."""
    found = None
    for sel_list, body in re.findall(r"([^{}]+)\{([^{}]*)\}", css):
        sels = [s.strip() for s in sel_list.split(",")]
        if selector in sels:
            m = re.findall(r"(?<![-\w])" + re.escape(prop) + r"\s*:\s*([^;]+);", body)
            if m:
                found = m[-1].strip()
    return found
```
